import org.java_websocket.WebSocket;
import org.java_websocket.handshake.ClientHandshake;
import org.java_websocket.server.WebSocketServer;
import java.net.InetSocketAddress;

public class ChatServer extends WebSocketServer {

// Pass a callback to the SessionManager, just like before
private final SessionManagerCallback sessionManager;

public ChatServer(int port, SessionManagerCallback sessionManager) {
super(new InetSocketAddress(port));
this.sessionManager = sessionManager;
}

@Override
public void onOpen(WebSocket conn, ClientHandshake handshake) {
System.out.println("[System] Incoming WebSocket connection from " + conn.getRemoteSocketAddress());
// Pass the new connection up to the ChatSessionManager to await the INIT JSON
sessionManager.onNewConnection(conn);
}

@Override
public void onClose(WebSocket conn, int code, String reason, boolean remote) {
sessionManager.onConnectionClosed(conn);
}

@Override
public void onMessage(WebSocket conn, String message) {
// The library reads the full JSON string for you. No more BufferedReader!
sessionManager.onMessageReceived(conn, message);
}

@Override
public void onError(WebSocket conn, Exception ex) {
System.err.println("[Error] Server error: " + ex.getMessage());
}

@Override
public void onStart() {
System.out.println("WebSocket Server started on port " + getPort());
}
}

import org.java_websocket.client.WebSocketClient;
import org.java_websocket.handshake.ServerHandshake;
import java.net.URI;

public class ChatClient extends WebSocketClient {

private final SessionManagerCallback sessionManager;

public ChatClient(URI serverUri, SessionManagerCallback sessionManager) {
super(serverUri);
this.sessionManager = sessionManager;
}

@Override
public void onOpen(ServerHandshake handshakedata) {
// We successfully connected to a peer.
// We pass 'this' (since ChatClient implements the WebSocket interface) to the manager
sessionManager.onNewConnection(this);
}

@Override
public void onMessage(String message) {
sessionManager.onMessageReceived(this, message);
}

@Override
public void onClose(int code, String reason, boolean remote) {
sessionManager.onConnectionClosed(this);
}

@Override
public void onError(Exception ex) {
System.err.println("[Error] Client error: " + ex.getMessage());
}
}

// Inside ChatSessionManager (Snippet)

// Instead of Map<String, PeerConnection>, you store Map<String, WebSocket>
private final Map<String, WebSocket> activeChats = new ConcurrentHashMap<>();

public void sendChatMessage(String peerId, String payload) {
WebSocket conn = activeChats.get(peerId);

if (conn != null && conn.isOpen()) {
Message msg = new Message(MessageType.MSG, myId, payload);
String json = JsonUtil.serialize(msg);

// This single line replaces the entire MessageSender class and BlockingQueue!
conn.send(json);
}
}

public void connectToPeer(String ip, int port) {
try {
URI uri = new URI("ws://" + ip + ":" + port);
ChatClient client = new ChatClient(uri, this);

// connect() is asynchronous. It will trigger onOpen() when successful
client.connect();
} catch (Exception e) {
ui.displayError("Invalid URI format.");
}
}








// 1. UPDATE THIS EXISTING METHOD
public void handleGroupInvite(Message msg) {
    GroupInviteData data = JsonUtil.deserializeInvite(msg.getPayload());
    if (data == null) return;

    ui.displaySystemMessage(msg.getSenderId() + " invited you to group: " + data.getGroupName());

    Group group = activeGroups.computeIfAbsent(data.getGroupId(), id -> new Group(id, data.getGroupName()));
    group.addMember(myId);
    group.addMember(msg.getSenderId());

    for (PeerAddress address : data.getCurrentMembers()) {
        group.addMember(address.getPeerId());
        registerKnownPeer(address.getPeerId(), address.getIp(), address.getPort());
        
        if (!address.getPeerId().equals(myId) && !sessionManager.isConnected(address.getPeerId())) {
            ui.displaySystemMessage("Mesh linking: Connecting to " + address.getPeerId() + "...");
            sessionManager.connectToPeer(address.getIp(), address.getPort());
        }
    }

    // NEW: The Announcement Logic
    // We run this in a background thread with a short delay to ensure the mesh 
    // handshakes (INIT/ACK) finish before we try to send messages.
    new Thread(() -> {
        try {
            Thread.sleep(1000); // 1 second delay
            PeerAddress myAddress = new PeerAddress(myId, myIp, myPort);
            String payload = JsonUtil.serializeAddress(myAddress);
            Message joinMsg = new Message(MessageType.GROUP_JOIN, myId, payload, data.getGroupId());

            // Announce to everyone in the group that we have arrived
            for (String peerId : group.getMemberIds()) {
                if (!peerId.equals(myId)) {
                    sessionManager.sendRawMessage(peerId, joinMsg);
                }
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }).start();
}

// 2. ADD THIS BRAND NEW METHOD
public void handleGroupJoin(Message msg) {
    Group group = activeGroups.get(msg.getGroupId());
    if (group != null) {
        // Deserialize the IP/Port info of the new member
        PeerAddress newMember = JsonUtil.deserializeAddress(msg.getPayload());
        
        if (newMember != null && !group.getMemberIds().contains(newMember.getPeerId())) {
            group.addMember(newMember.getPeerId());
            registerKnownPeer(newMember.getPeerId(), newMember.getIp(), newMember.getPort());
            
            ui.displaySystemMessage("[System] " + newMember.getPeerId() + " joined group '" + group.getGroupName() + "'");
        }
    }
}

// 3. UPDATE THIS METHOD (Adding the Self-Healing safety net)
public void handleIncomingGroupMessage(Message msg) {
    Group group = activeGroups.get(msg.getGroupId());
    String groupName = (group != null) ? group.getGroupName() : msg.getGroupId();
    
    // NEW: Self-Healing Logic. If we receive a message from someone we didn't know 
    // was in the group, add them immediately!
    if (group != null && !group.getMemberIds().contains(msg.getSenderId())) {
        group.addMember(msg.getSenderId());
        // Note: We don't have their IP/Port here, but at least we can now route messages back to them!
    }
    
    System.out.println("\n\u001B[35m[" + msg.getTimestamp() + "] [Group-" + groupName + "] " 
            + msg.getSenderId() + ": " + msg.getPayload() + "\u001B[0m");
}
